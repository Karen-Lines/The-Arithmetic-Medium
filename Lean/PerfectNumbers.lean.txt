import Mathlib

namespace MEADium.PerfectBlueprint

/-- The Collatz function. -/
def collatz (n : ℕ) : ℕ :=
  if n % 2 = 0 then n / 2 else 3 * n + 1

/-- The stopping time to 1. -/
def ST (n : ℕ) : ℕ :=
  if n = 1 then 0 
  else if n = 0 then 0
  else ST (collatz n) + 1
  termination_by n
